def test_print_permutations_short(capsys):
    worksheet.print_permutations([1, 2, 3])
    captured = set(capsys.readouterr().out.split("\n"))
    captured.remove("")
    expected_output = ["[1, 2, 3]", "[2, 1, 3]", "[3, 1, 2]", "[1, 3, 2]", "[2, 3, 1]", "[3, 2, 1]"]
    assert len(captured) == 6
    for i in expected_output:
        assert i in captured
